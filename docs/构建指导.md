docker run -itd -v $(pwd):/home/host_swap --name XXX_svf_env svf:v0
2. 使用vscode进入容器
3. 安装工具  apt update   apt-get install gdb   apt install libyaml-cpp-dev  apt-get install graphviz graphviz-dev
4. 进入/home/host_swap 运行 ./build.sh debug
5. 查看 .vscode/launch.json 中已有的配置
6. 在svf/lib/SABER/TaintChecker.cpp等文件中加断点并开始调试
